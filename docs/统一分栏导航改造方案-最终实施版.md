## 一、核心需求

### 1.1 需求清单（严格遵守）

| # | 需求项 | 说明 |
|---|--------|------|
| 1 | **统一分栏导航** | 所有页面都采用 Navigation 分栏导航方式，替代当前 URL 跳转 |
| 2 | **导航模式继承** | 子页面继承父页面的导航模式（独占/分栏） |
| 3 | **页面配置继承** | 子页面继承父页面的最大宽度、内边距等配置 |
| 4 | **首页不分栏** | 首页及其打开的页面不分栏，全屏显示，限制内容最大宽度 |
| 5 | **数据库分栏** | 数据库页面左侧展示条目列表，右侧显示条目详情，所有子页面都在右侧显示 |
| 6 | **页面复用** | RecentFiles 在首页打开时全屏，在设置中打开时在右侧显示 |
| 7 | **设置独立分栏** | 设置页面的分栏状态不能与数据库分栏掺和，使用独立导航栈 |
| 8 | **设置三级页面** | 设置的子页面使用统一样式的弹框展示，不注册到导航栈 |
| 9 | **弹框宽度自适应** | 屏幕宽度 > 700vp 时最大 700vp，≤ 700vp 时使用 100% 宽度 |
| 10 | **最小化修改** | 直接复用现有页面组件，现有设置子页面和条目页面零改动 |
| 11 | **参数传递** | 页面之间参数传递不能因改造导致参数获取不到 |
| 12 | **常量统一管理** | 所有硬编码数字、页面名称、样式参数抽取为公共常量 |
| 13 | **状态隔离与保持** | 不同导航栈之间的状态相互隔离，从一个导航栈返回另一个导航栈时，被返回的导航栈应保持离开时的状态（如数据库页面从设置返回后，应保持进入设置前的分组选择、滚动位置等状态） |

### 1.2 关键约束

```
┌─────────────────────────────────────────────────────────────────┐
│                        核心约束条件                              │
├─────────────────────────────────────────────────────────────────┤
│ 1. 导航模式由导航栈根页面决定，子页面自动继承                     │
│ 2. 设置页面使用独立导航栈，与数据库分栏状态互不影响               │
│ 3. 设置三级页面使用弹框，不注册到导航栈                          │
│ 4. 现有设置子页面和条目页面零改动，直接复用                       │
│ 5. 所有文本使用资源字符串，无硬编码中文                           │
│ 6. 所有常量统一管理，避免魔法数字                                │
│ 7. 导航栈状态相互隔离，切换导航栈时保持各自状态                   │
└─────────────────────────────────────────────────────────────────┘
```

```typescript
/**
 * 导航相关常量
 * 
 * 职责：管理导航组件的尺寸、阈值、根页面列表等配置
 */
export class NavigationConstants {
  /**
   * 导航栏宽度范围 [最小宽度, 最大宽度]
   * 用于 Navigation 组件的 navBarWidthRange 属性
   */
  static readonly NAV_BAR_WIDTH_RANGE: [number, number] = [240, 360];
  
  /**
   * 导航栏默认宽度（vp）
   */
  static readonly DEFAULT_NAV_BAR_WIDTH: number = 280;
  
  /**
   * 内容区最小宽度（vp）
   * 用于 Navigation 组件的 minContentWidth 属性
   */
  static readonly MIN_CONTENT_WIDTH: number = 400;
  
  /**
   * 分栏模式切换阈值（vp）
   * 屏幕宽度 >= 此值时自动切换为分栏模式
   * 用于 NavigationMode.Auto 的判断
   */
  static readonly SPLIT_MODE_THRESHOLD: number = 600;
  
  /**
   * 根页面列表
   * 这些页面会创建新的导航栈，决定导航模式
   */
  static readonly ROOT_PAGES: string[] = [
    'Index',         // 首页
    'DatabaseView',  // 数据库视图
    'Settings'       // 设置
  ];
  
  /**
   * 判断是否为根页面
   * 
   * @param pageName 页面名称
   * @returns true 表示是根页面
   */
  static isRootPage(pageName: string): boolean {
    return this.ROOT_PAGES.includes(pageName);
  }
}
```

## 四、导航模式继承机制

### 4.1 导航模式类型定义

**PageConfig.ets**：

```typescript
/**
 * 导航模式类型
 */
export enum NavigationModeType {
  /**
   * 独占模式（不分栏）
   * - 使用 NavigationMode.Stack
   * - 页面全屏显示
   * - 可以限制内容最大宽度
   * - 适用于：首页相关页面
   */
  EXCLUSIVE = 'exclusive',
  
  /**
   * 分栏模式
   * - 使用 NavigationMode.Auto 或 NavigationMode.Split
   * - 页面在分栏中显示
   * - 通常不限制内容宽度
   * - 适用于：数据库页面、设置页面
   */
  SPLIT = 'split',
}

/**
 * 页面配置接口
 */
export interface PageConfig {
  /**
   * 页面名称
   */
  name: string;
  
  /**
   * 导航模式类型
   */
  navigationMode: NavigationModeType;
  
  /**
   * 是否限制内容最大宽度
   * - true: 在独占模式下限制内容最大宽度
   * - false: 不限制宽度（分栏模式通常不限制）
   */
  constrainMaxWidth: boolean;
  
  /**
   * 内容最大宽度（vp）
   * 仅当 constrainMaxWidth 为 true 时有效
   */
  maxWidth?: number;
  
  /**
   * 内容区内边距（vp）
   */
  padding?: number;
}
```

### 4.2 导航模式继承规则

```
核心原则：导航模式由导航栈的根页面决定，子页面自动继承

导航栈根页面：决定导航模式
    └── 子页面：自动继承导航模式
        └── 孙页面：自动继承导航模式
```

```typescript
import { display } from '@kit.ArkUI';
import { UIConstants } from '../constants/UIConstants';

/**
 * UI 工具类
 */
export class UIUtils {
  /**
   * 计算内容最大宽度
   * 
   * 规则：
   * - 屏幕宽度 > 700vp：使用固定宽度 700vp
   * - 屏幕宽度 ≤ 700vp：使用 100% 宽度
   */
  static calculateMaxWidth(context: UIContext): { width: string | number; isPercentage: boolean } {
    try {
      const displayInfo = display.getDefaultDisplaySync();
      const screenWidthVp = context.px2vp(displayInfo.width);
      
      if (screenWidthVp > UIConstants.CONTENT_MAX_WIDTH_THRESHOLD) {
        return { width: UIConstants.CONTENT_MAX_WIDTH_THRESHOLD, isPercentage: false };
      } else {
        return { width: '100%', isPercentage: true };
      }
    } catch (error) {
      return { width: UIConstants.CONTENT_MAX_WIDTH_THRESHOLD, isPercentage: false };
    }
  }
  
  /**
   * 计算弹框宽度（自适应）
   * 复用 calculateMaxWidth 逻辑
   */
  static calculateDialogWidth(context: UIContext): { width: string | number; isPercentage: boolean } {
    return this.calculateMaxWidth(context);
  }
  
  /**
   * 判断是否使用百分比宽度
   */
  static usePercentageWidth(context: UIContext): boolean {
    const { isPercentage } = this.calculateMaxWidth(context);
    return isPercentage;
  }
  
  /**
   * 获取最大宽度常量
   */
  static getMaxWidth(): number {
    return UIConstants.CONTENT_MAX_WIDTH_THRESHOLD;
  }
}
```

### 7.1 NavigationManager - 导航管理器

```typescript
import { NavPathStack, NavigationMode } from '@kit.ArkUI';
import { PageConfigManager } from './PageConfigManager';
import { PageConfig, NavigationModeType } from './PageConfig';
import { NavigationConstants } from '../constants/NavigationConstants';

/**
 * 导航管理器
 * 
 * 职责：
 * 1. 管理导航栈
 * 2. 处理页面跳转
 * 3. 管理导航栈配置继承
 */
export class NavigationManager {
  private static instance: NavigationManager;
  private navPathStack: NavPathStack | null = null;
  private configManager: PageConfigManager = PageConfigManager.getInstance();
  
  /**
   * 当前导航栈的根页面
   */
  private currentRootPage: string = 'Index';
  
  /**
   * 导航历史
   */
  private navigationHistory: string[] = [];
  
  static getInstance(): NavigationManager {
    if (!NavigationManager.instance) {
      NavigationManager.instance = new NavigationManager();
    }
    return NavigationManager.instance;
  }
  
  /**
   * 设置导航栈
   */
  setNavPathStack(stack: NavPathStack): void {
    this.navPathStack = stack;
  }
  
  /**
   * 导航到页面
   * 
   * 关键逻辑：
   * 1. 判断是否是根页面
   * 2. 如果是根页面，设置导航栈配置
   * 3. 子页面自动继承导航栈配置
   */
  navigateTo(pageName: string, params?: Object): void {
    if (!this.navPathStack) {
      hilog.error(DOMAIN, TAG, 'NavPathStack not initialized');
      return;
    }
    
    const isRoot = this.configManager.isRootPage(pageName);
    
    if (isRoot) {
      // 设置新的导航栈根页面
      this.currentRootPage = pageName;
      
      // 获取根页面的配置
      const rootConfig = this.configManager.getPageConfig(pageName);
      if (rootConfig) {
        // 缓存导航栈配置
        this.configManager.setNavigationStackConfig(pageName, rootConfig);
      }
      
      // 重置导航历史
      this.navigationHistory = [pageName];
    } else {
      // 子页面：添加到导航历史
      this.navigationHistory.push(pageName);
    }
    
    // 执行导航
    this.navPathStack.pushPath({ name: pageName, param: params });
  }
  
  /**
   * 获取当前页面的有效配置（处理继承）
   */
  getCurrentPageConfig(pageName?: string): PageConfig {
    const currentPage = pageName || this.navigationHistory[this.navigationHistory.length - 1];
    return this.configManager.getEffectivePageConfig(currentPage, this.currentRootPage);
  }
  
  /**
   * 获取当前导航栈根页面
   */
  getCurrentRootPage(): string {
    return this.currentRootPage;
  }
  
  /**
   * 返回上一页
   */
  pop(): void {
    if (this.navPathStack && this.navPathStack.size() > 0) {
      this.navPathStack.pop();
      this.navigationHistory.pop();
      
      // 更新当前根页面
      if (this.navigationHistory.length > 0) {
        const lastPage = this.navigationHistory[this.navigationHistory.length - 1];
        if (this.configManager.isRootPage(lastPage)) {
          this.currentRootPage = lastPage;
        }
      }
    }
  }
  
  /**
   * 获取导航历史
   */
  getNavigationHistory(): string[] {
    return [...this.navigationHistory];
  }
  
  /**
   * 获取当前导航栈模式
   */
  getCurrentStackMode(): NavigationModeType {
    return this.getCurrentPageConfig().navigationMode;
  }
  
  /**
   * 转换为 NavigationMode
   */
  toNavigationMode(modeType: NavigationModeType): NavigationMode {
    return modeType === NavigationModeType.SPLIT 
      ? NavigationMode.Auto 
      : NavigationMode.Stack;
  }
}
```

### 7.5 Settings - 设置页面（弹框管理）

```typescript
import { SettingDialogWrapper } from '../../components/SettingDialogWrapper';
import { PageConstants } from '../../common/constants/PageConstants';

interface SettingItem {
  id: string;
  name: string;
  title: ResourceStr;
  isPageRoute?: boolean;
}

@Component
export struct Settings {
  @Prop navPathStack: NavPathStack;
  
  private settingItems: SettingItem[] = [
    { 
      id: 'security', 
      name: PageConstants.PAGE_SETTING_SECURITY, 
      title: $r('app.string.Settings_security') 
    },
    { 
      id: 'database', 
      name: PageConstants.PAGE_SETTING_DATABASE, 
      title: $r('app.string.Settings_database') 
    },
    { 
      id: 'application', 
      name: PageConstants.PAGE_SETTING_APPLICATION, 
      title: $r('app.string.Settings_application') 
    },
    { 
      id: 'about', 
      name: PageConstants.PAGE_ABOUT, 
      title: $r('app.string.Settings_about') 
    },
    { 
      id: 'recentFiles', 
      name: PageConstants.PAGE_RECENT_FILES, 
      title: $r('app.string.recent_files'), 
      isPageRoute: true 
    },
  ];
  
  private dialogController: CustomDialogController | null = null;
  
  build() {
    Column() {
      // 标题栏
      Row() {
        Image($r("app.media.ic_angle_left"))
          .onClick(() => this.navPathStack.pop())
        
        Text($r('app.string.Settings_title'))
      }
      
      // 设置列表
      List() {
        ForEach(this.settingItems, (item: SettingItem) => {
          ListItem() {
            Row() {
              Text(item.title)
              Image($r('app.media.ic_arrow_right'))
            }
            .onClick(() => this.handleItemClick(item))
          }
        })
      }
    }
  }
  
  private handleItemClick(item: SettingItem) {
    if (item.isPageRoute) {
      // 在右侧分栏显示
      this.navPathStack.pushPath({ name: item.name, param: { source: 'settings' } });
    } else {
      // 使用弹框展示
      this.showSettingDialog(item);
    }
  }
  
  private showSettingDialog(item: SettingItem) {
    this.dialogController = new CustomDialogController({
      builder: SettingDialogWrapper({
        title: item.title,
        context: this.getUIContext(),
        content: this.SettingDialogContentBuilder(item.name),
        onClose: () => this.dialogController?.close()
      }),
      alignment: DialogAlignment.Center,
      customStyle: true,
      autoCancel: true,
    });
    
    this.dialogController.open();
  }
  
  @Builder
  SettingDialogContentBuilder(settingName: string) {
    // 直接使用现有页面组件（最小改动）
    switch (settingName) {
      case PageConstants.PAGE_SETTING_SECURITY:
        SettingSecurity();
        break;
      case PageConstants.PAGE_SETTING_DATABASE:
        SettingDatabase();
        break;
      case PageConstants.PAGE_SETTING_APPLICATION:
        SettingApplication();
        break;
      case PageConstants.PAGE_ABOUT:
        About();
        break;
    }
  }
}
```

## 十一、总结

### 11.1 核心优势

| 优势 | 说明 |
|------|------|
| **导航模式继承** | 子页面自动继承父页面的导航模式 |
| **配置继承** | 子页面自动继承父页面的最大宽度等配置 |
| **统一管理** | 所有配置集中在 PageConfigManager |
| **常量统一** | 所有常量抽取为公共常量，易于维护 |
| **自适应弹框** | 设置弹框根据屏幕宽度自适应 |
| **最小改动** | 现有设置子页面和条目页面零改动 |
| **易于维护** | 修改根页面配置自动影响所有子页面 |
| **状态隔离** | 不同导航栈状态相互隔离，切换时保持各自状态 |

### 11.2 常量清单汇总

| 常量文件 | 常量类别 | 主要常量 |
|---------|---------|---------|
| NavigationConstants.ets | 导航配置 | NAV_BAR_WIDTH_RANGE, MIN_CONTENT_WIDTH, ROOT_PAGES |
| PageConstants.ets | 页面配置 | HOME_PAGE_MAX_WIDTH, 页面名称常量 |
| UIConstants.ets | UI通用 | CONTENT_MAX_WIDTH_THRESHOLD, CORNER_RADIUS_*, ICON_SIZE_* |
| SettingDialogConstants.ets | 弹框样式 | DIALOG_MAX_WIDTH, HEADER_HEIGHT, CORNER_RADIUS |
